    <script>
        // --- Telegram WebApp SDK Initialization ---
        const tg = window.Telegram ? window.Telegram.WebApp : null;

        if (tg) {
            tg.ready();
            tg.expand();
            // Enable closing confirmation if needed
            tg.enableClosingConfirmation();
        }

        // State variables
        let currentLang = 'ru';
        let cart = [];
        let balance = 250;

        // Translation Dictionary
        const translations = {
            ru: {
                catalog: "Каталог товаров",
                tasks: "Задания и Награды",
                profile_details: "Детали профиля",
                claim: "Забрать",
                invite: "Пригласить",
                nav_shop: "Магазин",
                nav_tasks: "Задания",
                nav_profile: "Профиль",
                language: "Язык системы",
                platform: "Платформа",
                test_popup: "Проверить Telegram Popup",
                send_to_bot: "Отправить данные боту",
                task_tg_channel: "Подписка на канал",
                task_invite: "Пригласить друга",
                task_daily: "Ежедневный бонус",
                add_to_cart: "В корзину",
                checkout: "Оформить заказ",
                cart_text: "Корзина: "
            },
            en: {
                catalog: "Product Catalog",
                tasks: "Tasks & Rewards",
                profile_details: "Profile Details",
                claim: "Claim",
                invite: "Invite",
                nav_shop: "Store",
                nav_tasks: "Tasks",
                nav_profile: "Profile",
                language: "System Language",
                platform: "Platform",
                test_popup: "Test Telegram Popup",
                send_to_bot: "Send Data to Bot",
                task_tg_channel: "Subscribe Channel",
                task_invite: "Invite a Friend",
                task_daily: "Daily Bonus",
                add_to_cart: "Add to Cart",
                checkout: "Checkout",
                cart_text: "Cart: "
            }
        };

        // Sample Catalog Data
        const products = [
            { id: 1, title: 'Telegram Premium (1 мес)', price: 299, icon: 'zap', color: 'from-amber-500 to-orange-600' },
            { id: 2, title: 'Telegram Stars (100 шт)', price: 199, icon: 'star', color: 'from-yellow-400 to-amber-500' },
            { id: 3, title: 'VIP Статус (30 дней)', price: 499, icon: 'shield-check', color: 'from-blue-500 to-indigo-600' },
            { id: 4, title: 'NFT Коллекционка #01', price: 999, icon: 'gem', color: 'from-purple-500 to-pink-600' }
        ];

        window.addEventListener('DOMContentLoaded', () => {
            initLucideIcons();
            loadUserData();
            renderProducts();
            setupMainButton();
        });

        function initLucideIcons() {
            if (window.lucide) {
                lucide.createIcons();
            }
        }

        function loadUserData() {
            if (tg && tg.initDataUnsafe && tg.initDataUnsafe.user) {
                const user = tg.initDataUnsafe.user;
                document.getElementById('user-name').textContent = `${user.first_name || ''} ${user.last_name || ''}`.trim() || 'Telegram User';
                document.getElementById('user-username').textContent = user.username ? `@${user.username}` : 'No username';
                
                // Avatar initials
                const initials = (user.first_name ? user.first_name[0] : 'U').toUpperCase();
                document.getElementById('user-avatar').textContent = initials;

                // Profile tab info
                document.getElementById('info-id').textContent = user.id || 'N/A';
                document.getElementById('info-lang').textContent = user.language_code || 'ru';
                document.getElementById('info-platform').textContent = tg.platform || 'mobile';
            }
        }

        function renderProducts() {
            const grid = document.getElementById('products-grid');
            grid.innerHTML = products.map(product => `
                <div class="glass-card rounded-2xl p-3.5 flex flex-col justify-between border border-slate-800 hover:border-slate-700 transition">
                    <div class="space-y-2">
                        <div class="w-10 h-10 rounded-xl bg-gradient-to-tr ${product.color} flex items-center justify-center text-white shadow-md">
                            <i data-lucide="${product.icon}" class="w-5 h-5"></i>
                        </div>
                        <div>
                            <h4 class="text-xs font-semibold leading-snug line-clamp-2">${product.title}</h4>
                            <p class="text-xs font-bold text-blue-400 mt-1">${product.price} ₽</p>
                        </div>
                    </div>
                    <button onclick="addToCart(${product.id})" class="mt-3 w-full bg-slate-800 hover:bg-slate-700 text-blue-400 text-xs py-1.5 rounded-lg border border-slate-700 transition font-medium flex items-center justify-center gap-1">
                        <i data-lucide="plus" class="w-3.5 h-3.5"></i>
                        <span data-i18n="add_to_cart">${translations[currentLang].add_to_cart}</span>
                    </button>
                </div>
            `).join('');
            initLucideIcons();
        }

        function triggerHaptic(type = 'impact', style = 'light') {
            if (!tg || !tg.HapticFeedback) return;
            if (type === 'impact') {
                tg.HapticFeedback.impactOccurred(style);
            } else if (type === 'notification') {
                tg.HapticFeedback.notificationOccurred(style);
            } else if (type === 'selection') {
                tg.HapticFeedback.selectionChanged();
            }
        }

        // Cart functionality
        function addToCart(productId) {
            triggerHaptic('impact', 'light');
            const product = products.find(p => p.id === productId);
            if (product) {
                cart.push(product);
                updateCartUI();
            }
        }

        function updateCartUI() {
            document.getElementById('cart-count').textContent = cart.length;
            
            // Telegram MainButton control
            if (tg && tg.MainButton) {
                if (cart.length > 0) {
                    const totalPrice = cart.reduce((sum, item) => sum + item.price, 0);
                    tg.MainButton.setText(`${translations[currentLang].checkout} (${totalPrice} ₽)`);
                    tg.MainButton.show();
                } else {
                    tg.MainButton.hide();
                }
            }
        }

        function setupMainButton() {
            if (tg && tg.MainButton) {
                tg.MainButton.setParams({
                    color: '#2563eb',
                    text_color: '#ffffff'
                });
                tg.MainButton.onClick(() => {
                    triggerHaptic('notification', 'success');
                    sendCartData();
                });
            }
        }

        function sendCartData() {
            if (tg) {
                tg.sendData(JSON.stringify({
                    action: 'checkout',
                    items: cart,
                    total: cart.reduce((sum, item) => sum + item.price, 0)
                }));
            }
        }

        // Tasks & Rewards functionality
        function completeTask(btn, points) {
            triggerHaptic('notification', 'success');
            balance += points;
            document.getElementById('user-balance').textContent = `Баланс: ${balance} PTS`;
            btn.disabled = true;
            btn.className = "bg-slate-800 text-slate-500 text-xs px-3 py-1.5 rounded-lg font-medium cursor-not-allowed";
            btn.textContent = "Готово";
        }

        function inviteFriend() {
            triggerHaptic('impact', 'medium');
            if (tg) {
                tg.openTelegramLink('https://t.me/share/url?url=https://t.me/your_bot_username&text=Присоединяйся%20к%20Mini%20App!');
            }
        }

        // Popup demo
        function showPopup() {
            if (tg && tg.showPopup) {
                tg.showPopup({
                    title: 'Telegram Native Popup',
                    message: 'Это интерактивное системное окно Telegram WebApp!',
                    buttons: [{ type: 'ok' }]
                });
            } else {
                alert('Telegram WebApp Popup API не поддерживается в браузере.');
            }
        }

        function sendDataToBot() {
            if (tg) {
                tg.sendData(JSON.stringify({ action: 'ping', timestamp: Date.now() }));
            }
        }

        function switchTab(tabName) {
            triggerHaptic('selection');

            // Hide all tabs
            ['shop', 'tasks', 'profile'].forEach(name => {
                const section = document.getElementById(`tab-${name}`);
                const navBtn = document.getElementById(`nav-${name}`);
                
                if (name === tabName) {
                    section.classList.remove('hidden');
                    navBtn.classList.add('active-tab');
                    navBtn.classList.remove('text-slate-400');
                } else {
                    section.classList.add('hidden');
                    navBtn.classList.remove('active-tab');
                    navBtn.classList.add('text-slate-400');
                }
            });
        }

        function toggleTheme() {
            triggerHaptic('impact', 'light');
            const body = document.body;
            const themeIcon = document.getElementById('theme-icon');
            
            if (body.classList.contains('light')) {
                body.classList.remove('light');
                themeIcon.setAttribute('data-lucide', 'moon');
            } else {
                body.classList.add('light');
                themeIcon.setAttribute('data-lucide', 'sun');
            }
            initLucideIcons();
        }

        function toggleLanguage() {
            triggerHaptic('impact', 'light');
            currentLang = currentLang === 'ru' ? 'en' : 'ru';
            document.getElementById('lang-indicator').textContent = currentLang.toUpperCase();

            // Update DOM translations
            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (translations[currentLang][key]) {
                    el.textContent = translations[currentLang][key];
                }
            });

            // Re-render components with translated texts
            renderProducts();
            updateCartUI();
        }
    </script>
